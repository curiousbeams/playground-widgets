:::{marimo} python
import numpy as np
import matplotlib.pyplot as plt
import marimo as mo
:::

:::{marimo} python
:echo: true
:output: true

phase_slider = mo.ui.slider(start=0.0, stop=2 * np.pi, step=0.1, value=0.0, label="Time Phase (phi)")
:::

:::{marimo} python
:echo: true

phi = phase_slider.value

x = np.linspace(-5, 5, 200)
y = np.linspace(-5, 5, 200)

xa, ya = np.meshgrid(x, y)
za = np.sin(xa - phi) * np.cos(ya - phi)
:::

:::{marimo} python
:echo: false
:output: false

fig, ax = plt.subplots()
ax.imshow(za,cmap='RdBu',extent=[-5,5,-5,5],origin='lower')
None
:::

:::{marimo} python
mo.vstack([phase_slider,fig],align="center")
